cmake_minimum_required(VERSION 3.21)
project(ptoolkit_editor VERSION 0.1.0 LANGUAGES CXX)

if(NOT TARGET ftxui::component)
  find_package(ftxui CONFIG REQUIRED)
endif()
if(ftxui_VERSION)
  set(_ftxui_version "${ftxui_VERSION}")
else()
  get_target_property(_ftxui_source ftxui::component SOURCE_DIR)
  get_directory_property(_ftxui_version DIRECTORY "${_ftxui_source}"
                         DEFINITION ftxui_VERSION)
endif()
if(NOT _ftxui_version OR _ftxui_version VERSION_LESS 7.0.3)
  message(FATAL_ERROR "ptoolkit_editor requires a known FTXUI version >= 7.0.3")
endif()

add_library(ptoolkit_editor STATIC document.cc document_multiselect.cc json_highlight.cc editor.cc editor_input.cc clipboard.cc)
add_library(ptoolkit::editor ALIAS ptoolkit_editor)
target_compile_features(ptoolkit_editor PUBLIC cxx_std_23)
target_include_directories(ptoolkit_editor PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(ptoolkit_editor PUBLIC ftxui::component ftxui::dom ftxui::screen)
if(WIN32)
  target_link_libraries(ptoolkit_editor PRIVATE user32)
endif()

option(PTOOLKIT_EDITOR_BUILD_TESTS "Build independent editor tests" OFF)
option(PTOOLKIT_EDITOR_BUILD_EXAMPLE "Build standalone JSON editor example" OFF)
if(PTOOLKIT_EDITOR_BUILD_EXAMPLE)
  add_executable(ptoolkit_editor_demo examples/editor_demo.cc)
  target_link_libraries(ptoolkit_editor_demo PRIVATE ptoolkit::editor)
  if(WIN32)
    add_executable(ptoolkit_editor_mouse_probe tests/mouse_probe.cc)
    target_link_libraries(ptoolkit_editor_mouse_probe PRIVATE ptoolkit::editor)
    add_executable(ptoolkit_editor_multiselect_probe tests/multiselect_probe.cc)
    target_link_libraries(ptoolkit_editor_multiselect_probe PRIVATE ptoolkit::editor)
  endif()
endif()
if(PTOOLKIT_EDITOR_BUILD_TESTS)
  enable_testing()
  add_executable(ptoolkit_editor_tests tests/editor_tests.cc)
  target_link_libraries(ptoolkit_editor_tests PRIVATE ptoolkit::editor)
  add_test(NAME ptoolkit_editor_tests COMMAND ptoolkit_editor_tests)
  if(WIN32)
    add_test(NAME ptoolkit_editor_clipboard_tests COMMAND ptoolkit_editor_tests --system-clipboard-test)
  endif()
endif()
